# Capture the exact source state represented by this package build.
find_package(Git REQUIRED)
execute_process(
    COMMAND "${GIT_EXECUTABLE}" rev-parse --verify HEAD
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    RESULT_VARIABLE SHMEA_GIT_RESULT
    OUTPUT_VARIABLE SHMEA_SOURCE_REVISION
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET)
if(NOT SHMEA_GIT_RESULT EQUAL 0)
    message(FATAL_ERROR "Unable to determine shmea source revision")
endif()
execute_process(
    COMMAND "${GIT_EXECUTABLE}" status --porcelain=v1 --untracked-files=all
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    RESULT_VARIABLE SHMEA_STATUS_RESULT
    OUTPUT_VARIABLE SHMEA_SOURCE_STATUS
    ERROR_QUIET)
if(NOT SHMEA_STATUS_RESULT EQUAL 0)
    message(FATAL_ERROR "Unable to determine shmea source dirty state")
endif()
if(SHMEA_SOURCE_STATUS STREQUAL "")
    set(SHMEA_SOURCE_DIRTY FALSE)
else()
    set(SHMEA_SOURCE_DIRTY TRUE)
endif()
execute_process(
    COMMAND "${GIT_EXECUTABLE}" diff --binary --full-index --no-ext-diff HEAD -- .
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    RESULT_VARIABLE SHMEA_DIFF_RESULT
    OUTPUT_VARIABLE SHMEA_SOURCE_DIFF
    ERROR_QUIET)
if(NOT SHMEA_DIFF_RESULT EQUAL 0)
    message(FATAL_ERROR "Unable to capture shmea source diff")
endif()
string(SHA256 SHMEA_SOURCE_DIFF_SHA256 "${SHMEA_SOURCE_DIFF}")
set(SHMEA_SOURCE_ID "${SHMEA_SOURCE_REVISION}")
if(SHMEA_SOURCE_DIRTY)
    string(APPEND SHMEA_SOURCE_ID "-dirty.${SHMEA_SOURCE_DIFF_SHA256}")
endif()

    INCLUDES DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}

    LIBRARY  DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME  DESTINATION ${CMAKE_INSTALL_BINDIR})

# Export the complete target set and pair it with a config that records the
# source revision and working-tree diff used to build the package.
set(shmea_INSTALL_CMAKEDIR "${CMAKE_INSTALL_DATADIR}/shmea/cmake")

# The legacy subdirectories each create their own build-tree export. Load them
# before the root target so all of shmea's link-interface targets are defined.
export(TARGETS ${PROJECT_NAME}
    FILE "${CMAKE_CURRENT_BINARY_DIR}/shmeaBuildTargets.cmake")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shmeaConfig.cmake"
"# Auto-generated by ShmeaDB/CMakeLists.txt\n"
"set(shmea_VERSION \"${G_VERSION_MAJOR}.${G_VERSION_MINOR}\")\n"
"set(shmea_SOURCE_REVISION \"${SHMEA_SOURCE_REVISION}\")\n"
"set(shmea_SOURCE_DIRTY ${SHMEA_SOURCE_DIRTY})\n"
"set(shmea_SOURCE_DIFF_SHA256 \"${SHMEA_SOURCE_DIFF_SHA256}\")\n"
"set(shmea_SOURCE_ID \"${SHMEA_SOURCE_ID}\")\n"
"include(\"\${CMAKE_CURRENT_LIST_DIR}/Backend/Database/shmeaConfig.cmake\")\n"
"include(\"\${CMAKE_CURRENT_LIST_DIR}/Backend/Plotter/shmeaConfig.cmake\")\n"
"include(\"\${CMAKE_CURRENT_LIST_DIR}/Backend/Networking/shmeaConfig.cmake\")\n"
"include(\"\${CMAKE_CURRENT_LIST_DIR}/shmeaBuildTargets.cmake\")\n")

# The install export is consolidated, so its config has one target include.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shmeaInstallConfig.cmake"
"# Auto-generated by ShmeaDB/CMakeLists.txt\n"
"set(shmea_VERSION \"${G_VERSION_MAJOR}.${G_VERSION_MINOR}\")\n"
"set(shmea_SOURCE_REVISION \"${SHMEA_SOURCE_REVISION}\")\n"
"set(shmea_SOURCE_DIRTY ${SHMEA_SOURCE_DIRTY})\n"
"set(shmea_SOURCE_DIFF_SHA256 \"${SHMEA_SOURCE_DIFF_SHA256}\")\n"
"set(shmea_SOURCE_ID \"${SHMEA_SOURCE_ID}\")\n"
"include(\"\${CMAKE_CURRENT_LIST_DIR}/shmeaTargets.cmake\")\n")

    RENAME shmeaConfig.cmake)

    DESTINATION "${shmea_INSTALL_CMAKEDIR}")
